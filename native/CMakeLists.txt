cmake_minimum_required(VERSION 3.20)
project(paper_native_workers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_SOURCE_DIR "" CACHE PATH "Pinned llama.cpp checkout")
set(LLAMA_LIBRARY_DIR "" CACHE PATH "Built llama.cpp shared libraries")
option(BUILD_PRECISION "Build partitioned precision workers and tests" ON)
option(BUILD_PREFILL "Build prefill and attention workers" OFF)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned checkout")
endif()
include_directories(include "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include")
set(CMAKE_BUILD_RPATH "${LLAMA_LIBRARY_DIR}")
foreach(component llama ggml ggml-base ggml-cpu)
    find_library(${component}_LIBRARY NAMES ${component} PATHS "${LLAMA_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
    list(APPEND RUNTIME_LIBRARIES "${${component}_LIBRARY}")
endforeach()
if(BUILD_PRECISION)
    enable_testing()
    add_library(research-executor STATIC llama_cpu_executor.cpp tensor.cpp)
    target_link_libraries(research-executor PUBLIC ${RUNTIME_LIBRARIES})
    add_executable(qwen-route-worker route_worker.cpp)
    target_link_libraries(qwen-route-worker PRIVATE research-executor)
    add_executable(qwen-release-worker route_worker.cpp)
    target_compile_definitions(qwen-release-worker PRIVATE DLI_RELEASE_WEIGHTS=1)
    target_link_libraries(qwen-release-worker PRIVATE research-executor)
    add_executable(u3-microbatch2-worker legacy_decode.cpp)
    target_link_libraries(u3-microbatch2-worker PRIVATE ${RUNTIME_LIBRARIES})
    add_executable(test-qwen-math tests/test_qwen_math.cpp)
    target_include_directories(test-qwen-math PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(test-qwen-math PRIVATE -UNDEBUG)
    add_test(NAME qwen-math COMMAND test-qwen-math)
    add_executable(test-qwen-attention tests/test_qwen_attention.cpp)
    target_link_libraries(test-qwen-attention PRIVATE research-executor)
    target_compile_options(test-qwen-attention PRIVATE -UNDEBUG)
    add_test(NAME qwen-attention COMMAND test-qwen-attention)
endif()
if(BUILD_PREFILL)
    add_executable(attention-worker prefill_worker.cpp)
    target_link_libraries(attention-worker PRIVATE ${RUNTIME_LIBRARIES})
    add_executable(attention-reference attention_reference.cpp)
    target_link_libraries(attention-reference PRIVATE ${RUNTIME_LIBRARIES})
endif()
